# Checkpoint 4f: torch + sentence-transformers removed. A continuous
# psutil RSS trace showed baseline memory climbing past 700MB within 15
# seconds of process start, before any request or model load - the cost
# is torch/sentence-transformers/langchain's own import and runtime
# footprint, not model weight size (Checkpoint 4e's fix, swapping to a
# smaller model alone, addressed the wrong variable). rag/embedder.py
# runs the same logical embedding model via onnxruntime instead, which
# has no torch dependency; tokenizers handles tokenization (pure Rust
# bindings, no torch/transformers dependency either); huggingface-hub
# fetches the ONNX model/tokenizer files, the same lazy-download-and-
# cache mechanism sentence-transformers used internally, just without
# the rest of that framework attached.
onnxruntime==1.29.0
tokenizers==0.23.2
huggingface-hub==1.29.0

# Checkpoint 7: PDF text extraction for user uploads. pypdf is NOT a
# dependency of langchain-community (checked: it is an optional extra of
# that package's PDF loaders, and was not installed here), so this is a
# real new dependency rather than a pin on something already present.
# Deliberately not pdfplumber or unstructured, and never PyMuPDF - that
# one is AGPL-licensed and must not enter this repo.
pypdf==6.17.0
# pypdf's "fonts" extra. Without it, _cmap.py logs a warning per CFF
# Type1 font and returns early, skipping that font's encoding table -
# which is what the repeated pypdf._cmap lines in Render's logs are. Note
# the direction: missing fontTools makes pypdf do LESS work, not more, so
# this is an extraction-QUALITY fix (unresolved encodings come back as
# garbled text, which then gets embedded and retrieved), not a speed one.
fonttools==4.65.0
